cmake_minimum_required(VERSION 3.25)
project(Caissa VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CAISSA_SANITIZE "Enable address and undefined behavior instrumentation" OFF)
option(CAISSA_TSAN "Enable thread instrumentation separately" OFF)
if(CAISSA_SANITIZE AND CAISSA_TSAN)
  message(FATAL_ERROR "Address and thread sanitizers require separate builds")
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(CAISSA_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
if(CAISSA_TSAN)
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
endif()
set(CAISSA_NETWORK_WIDTH "64" CACHE STRING "Active sparse-transformer width: 32, 64 or 128")
if(NOT CAISSA_NETWORK_WIDTH MATCHES "^(32|64|128)$")
  message(FATAL_ERROR "CAISSA_NETWORK_WIDTH must be 32, 64 or 128")
endif()
option(CAISSA_LTO "Enable supported release interprocedural optimization" ON)
if(CAISSA_LTO AND NOT CAISSA_SANITIZE AND NOT CAISSA_TSAN)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT CAISSA_IPO_SUPPORTED OUTPUT CAISSA_IPO_DIAGNOSTIC)
  if(CAISSA_IPO_SUPPORTED)
    set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
  else()
    message(STATUS "Release IPO unavailable: ${CAISSA_IPO_DIAGNOSTIC}")
  endif()
endif()
find_package(Threads REQUIRED)
add_library(caissa_core src/version.cpp src/attacks.cpp src/position.cpp src/position_line.cpp src/movegen.cpp src/reference.cpp src/rules.cpp src/evaluation.cpp src/table.cpp src/search.cpp src/uci.cpp src/selfplay.cpp src/network.cpp src/network_kernel.cpp src/notation.cpp src/serialization.cpp src/protocol_io.cpp)
target_include_directories(caissa_core PUBLIC include)
target_compile_definitions(caissa_core PUBLIC CAISSA_NETWORK_WIDTH=${CAISSA_NETWORK_WIDTH})
target_link_libraries(caissa_core PUBLIC Threads::Threads)
add_executable(caissa apps/main.cpp)
target_link_libraries(caissa PRIVATE caissa_core)
enable_testing()
add_test(NAME version COMMAND caissa --version)
install(TARGETS caissa RUNTIME DESTINATION bin)

add_executable(core_test tests/core_test.cpp)
target_link_libraries(core_test PRIVATE caissa_core)
add_test(NAME core COMMAND core_test)
add_executable(position_test tests/position_test.cpp)
target_link_libraries(position_test PRIVATE caissa_core)
add_test(NAME position COMMAND position_test)

add_executable(rules_test tests/rules_test.cpp)
target_link_libraries(rules_test PRIVATE caissa_core)
add_test(NAME rules COMMAND rules_test)
add_executable(evaluation_test tests/evaluation_test.cpp)
target_link_libraries(evaluation_test PRIVATE caissa_core)
add_test(NAME evaluation COMMAND evaluation_test)

add_executable(table_test tests/table_test.cpp)
target_link_libraries(table_test PRIVATE caissa_core)
add_test(NAME table COMMAND table_test)

add_executable(search_test tests/search_test.cpp)
target_link_libraries(search_test PRIVATE caissa_core)
add_test(NAME search COMMAND search_test)

add_executable(network_test tests/network_test.cpp)
target_link_libraries(network_test PRIVATE caissa_core)
add_executable(notation_test tests/notation_test.cpp)
target_link_libraries(notation_test PRIVATE caissa_core)
add_test(NAME notation COMMAND notation_test)
add_executable(uci_test tests/uci_test.cpp)
target_link_libraries(uci_test PRIVATE caissa_core)
add_test(NAME uci COMMAND uci_test)
add_executable(position_line_test tests/position_line_test.cpp)
target_link_libraries(position_line_test PRIVATE caissa_core)
add_test(NAME position_line COMMAND position_line_test)
set_tests_properties(position_line PROPERTIES TIMEOUT 30)
add_executable(kernel_test tests/kernel_test.cpp)
target_link_libraries(kernel_test PRIVATE caissa_core)
add_test(NAME kernels COMMAND kernel_test)
add_executable(dense_kernel_test tests/dense_kernel_test.cpp)
target_link_libraries(dense_kernel_test PRIVATE caissa_core)
add_test(NAME dense_kernels COMMAND dense_kernel_test)
add_executable(dense_kernel_bench tests/dense_kernel_bench.cpp)
target_link_libraries(dense_kernel_bench PRIVATE caissa_core)

add_executable(neural_bounds_test tests/neural_bounds_test.cpp)
target_link_libraries(neural_bounds_test PRIVATE caissa_core)
if(CAISSA_NETWORK_WIDTH STREQUAL "64")
  add_test(NAME neural_bounds COMMAND neural_bounds_test ${CMAKE_SOURCE_DIR}/models/residual-prefix-diagnostic.cai)
else()
  add_test(NAME neural_bounds COMMAND neural_bounds_test)
endif()
